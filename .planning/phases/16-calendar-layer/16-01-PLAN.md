    // === Phase 16 / CAL-16-02 — recurring watchparty series (top-level) ===
    // Top-level collection per Phase 30 architecture. memberUids-gated read.
    // Creator-only mutations. titleType immutable post-create. Soft-delete only.
    match /watchpartySeries/{seriesId} {
      // READ: any member listed in series.memberUids
      allow read: if signedIn() && request.auth.uid in resource.data.memberUids;

      // CREATE: creator must be in own memberUids; familyCode must be one the
      // caller belongs to (matches Phase 30 top-level wp create gate).
      allow create: if signedIn()
        && request.resource.data.createdByUid == uid()
        && request.resource.data.createdBy is string
        && request.resource.data.createdBy.matches('^m_[A-Za-z0-9_-]+$')
        && request.auth.uid in request.resource.data.memberUids
        && request.resource.data.familyCode is string
        && exists(/databases/$(database)/documents/users/$(uid())/groups/$(request.resource.data.familyCode))
        && request.resource.data.status == 'active'
        && request.resource.data.titleType in ['tv', 'untitled']
        && request.resource.data.daysOfWeek is list
        && request.resource.data.daysOfWeek.size() >= 1
        && request.resource.data.daysOfWeek.size() <= 7
        && request.resource.data.timeOfDay is string
        && request.resource.data.timeOfDay.matches('^[0-2][0-9]:[0-5][0-9]$')
        && request.resource.data.timezone is string
        && request.resource.data.timezone.size() > 0
        && request.resource.data.actingUid == uid()
        && request.resource.data.memberId is string
        && request.resource.data.memberId.matches('^m_[A-Za-z0-9_-]+$');

      // UPDATE: only the creator can edit. Allowlist of mutable fields.
      // titleType cannot change post-create (CONTEXT requirement).
      // CF (admin SDK) bypasses rules for nextFireAt / lastFiredAt advancement.
      allow update: if signedIn()
        && resource.data.createdByUid == request.auth.uid
        && request.resource.data.actingUid == request.auth.uid
        && request.resource.data.memberId is string
        && request.resource.data.memberId.matches('^m_[A-Za-z0-9_-]+$')
        && (
          !('managedMemberId' in request.resource.data)
          || (
            request.resource.data.managedMemberId is string
            && request.resource.data.managedMemberId.matches('^m_[A-Za-z0-9_-]+$')
          )
        )
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'daysOfWeek', 'timeOfDay', 'timezone', 'memberUids',
          'titleId', 'titleName',
          'status', 'pausedAt', 'endedAt', 'nextFireAt', 'lastFiredAt',
          'actingUid', 'managedMemberId', 'memberId', 'memberName'
        ])
        && resource.data.titleType == request.resource.data.titleType;

      // DELETE: denied — use status='ended' for soft-delete (mirrors Phase 30 wps).
      allow delete: if false;
    }

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client (browser PWA) → Firestore | untrusted: any signed-in user can attempt arbitrary writes; rules are the only enforcement layer |
| admin SDK (Cloud Function) → Firestore | trusted: bypasses rules; CF must self-validate inputs |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-16-01 | Spoofing | watchpartySeries create | mitigate | Rule requires `request.resource.data.createdByUid == uid()` + caller-is-in-familyCode-groups gate (mirrors Phase 30 hostUid/hostFamilyCode pattern) |
| T-16-04 | Information Disclosure | watchpartySeries read | mitigate | Read gated on `request.auth.uid in resource.data.memberUids` (verbatim Phase 30 top-level wp pattern at firestore.rules:161) |
| T-16-06 | Elevation of Privilege | watchpartySeries update by non-creator | mitigate | Update requires `resource.data.createdByUid == request.auth.uid` (creator-only); attribution echo required on every write |
| T-16-09 (new) | Tampering | titleType change post-create | mitigate | Update rule asserts `resource.data.titleType == request.resource.data.titleType` (immutability invariant per CONTEXT lock) |
