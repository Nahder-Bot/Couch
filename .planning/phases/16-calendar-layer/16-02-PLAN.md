```js
'use strict';

// === Phase 16 / CAL-16-12 — computeNextFireAt: DST-aware cadence helper ===
// Pure CJS function — no admin, no Firestore. Importable from:
//   - queuenight/functions/src/watchpartySeriesTick.js (server CF)
//   - scripts/smoke-series-cadence-compute.cjs (Node smoke test)
//
// Why no luxon / no tz library: CLAUDE.md "no bundler" rule.
// Intl-based pattern is verified at queuenight/functions/index.js:50-75 (isInQuietHours).
//
// DST correctness: getTimezoneOffsetForDate derives the offset for the SPECIFIC instant,
// so 'Sun 2:30 AM America/Los_Angeles' on a spring-forward Sunday resolves to 3:30 PDT
// (the wall-clock 2:30 simply doesn't exist; the Date.UTC math walks past it).

const WEEKDAY_NAMES = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
const DAY_MS = 24 * 60 * 60 * 1000;

/**
 * Returns the tz offset (ms) for the given instant in the given IANA timezone.
 * Positive means west of UTC (e.g. America/Los_Angeles PST = +8*3600*1000).
 *
 * Implementation: format the instant in the target tz, then format it in UTC,
 * reconstruct both wall-clock instants as Date.UTC values, take the difference.
 * This is DST-correct because we ask Intl for the actual instant's parts.
 */
function getTimezoneOffsetForDate(utcMs, timezone) {
  const date = new Date(utcMs);
  const tzParts = new Intl.DateTimeFormat('en-US', {
    timeZone: timezone,
    year: 'numeric', month: '2-digit', day: '2-digit',
    hour: '2-digit', minute: '2-digit', second: '2-digit',
    hour12: false
  }).formatToParts(date);
  const get = (type) => Number(tzParts.find(p => p.type === type).value);
  // Intl returns hour=24 for midnight in some locales; normalize to 0.
  const hour = get('hour') === 24 ? 0 : get('hour');
  const tzWallClockUtcMs = Date.UTC(
    get('year'), get('month') - 1, get('day'),
    hour, get('minute'), get('second')
  );
  // offsetMs = UTC clock - target-tz wall clock (positive west of UTC).
  return utcMs - tzWallClockUtcMs;
}

/**
 * @param {number[]} daysOfWeek  0=Sun .. 6=Sat (e.g. [1,3] for Mon+Wed). MUST be non-empty.
 * @param {string} timeOfDayHHMM 'HH:MM' 24h, in the given timezone's wall clock
 * @param {string} timezone      IANA name (e.g. 'America/Los_Angeles')
 * @param {number} [now]         epoch ms; default Date.now()
 * @returns {number | null}      epoch ms of next fire (or null if daysOfWeek empty)
 */
function computeNextFireAt(daysOfWeek, timeOfDayHHMM, timezone, now) {
  if (!Array.isArray(daysOfWeek) || daysOfWeek.length === 0) return null;
  if (typeof timeOfDayHHMM !== 'string' || !/^[0-2][0-9]:[0-5][0-9]$/.test(timeOfDayHHMM)) return null;
  if (typeof timezone !== 'string' || !timezone) return null;
  const t = (typeof now === 'number') ? now : Date.now();
  const [hh, mm] = timeOfDayHHMM.split(':').map(Number);

  // Walk forward 1 day at a time from `now` (8 day window covers any cadence + same-day-past wrap).
  for (let dayOffset = 0; dayOffset < 8; dayOffset++) {
    const probe = new Date(t + dayOffset * DAY_MS);
    // Get the weekday IN THE TARGET TZ (NOT UTC) using Intl.
    const parts = new Intl.DateTimeFormat('en-US', {
      timeZone: timezone,
      weekday: 'short',
      year: 'numeric', month: '2-digit', day: '2-digit'
    }).formatToParts(probe);
    const weekdayName = parts.find(p => p.type === 'weekday').value;
    const dow = WEEKDAY_NAMES.indexOf(weekdayName);
    if (dow < 0 || !daysOfWeek.includes(dow)) continue;

    const y = Number(parts.find(p => p.type === 'year').value);
    const mo = Number(parts.find(p => p.type === 'month').value);
    const d = Number(parts.find(p => p.type === 'day').value);

    // Construct the wall-clock instant `y-mo-d hh:mm` AS IF it were UTC, then
    // adjust by the tz offset for that instant to get the real UTC epoch ms.
    const utcGuess = Date.UTC(y, mo - 1, d, hh, mm);
    const tzOffset = getTimezoneOffsetForDate(utcGuess, timezone);
    const fireAt = utcGuess + tzOffset;

    if (fireAt > t) return fireAt;
    // else: same-day but past — fall through to next day
  }
  return null;
}

module.exports = { computeNextFireAt, getTimezoneOffsetForDate, WEEKDAY_NAMES };
```

NOTE on the offset sign — the formula `fireAt = utcGuess + tzOffset` works because `tzOffset` is defined as `utcMs - tzWallClockUtcMs` (positive west of UTC). For LA (PST = UTC-8), tzOffset = +8h, and a wall-clock `Date.UTC(y,m,d,20,0)` represents "20:00 UTC" which we want to shift to "20:00 PST = 04:00 UTC next day" so we add 8h. Correct.
