**Step 3: Sharp-install + optimize the 5 raw captures from the user's staging path.**

```bash
RAW_DIR="<path-from-resume-signal>"  # e.g. /c/Users/nahde/Desktop/phase-31-raw
mkdir -p /tmp/img-opt && cd /tmp/img-opt && npm install --silent sharp 2>&1 | tail -3
cd "$REPO_ROOT"  # back to couch repo root

node -e '
const sharp = require("/tmp/img-opt/node_modules/sharp");
const path = require("path");
const fs = require("fs");
const RAW = process.env.RAW_DIR;
const targets = [
  { src: "tonight-couch-viz.png",   dst: "marketing/tonight-couch-viz.png" },
  { src: "watchparty-wait-up.png",  dst: "marketing/watchparty-wait-up.png" },
  { src: "couch-groups-roster.png", dst: "marketing/couch-groups-roster.png" },
  { src: "tonight-pickup.png",      dst: "marketing/tonight-pickup.png" },
  { src: "pickem-leaderboard.png",  dst: "marketing/pickem-leaderboard.png" },
];
(async () => {
  for (const t of targets) {
    const srcPath = path.join(RAW, t.src);
    if (!fs.existsSync(srcPath)) {
      console.warn("SKIP (raw missing): " + srcPath);
      continue;
    }
    const stat = fs.statSync(srcPath);
    // For PNG inputs ≥ 400KB, run palette + compressionLevel 9 per Phase 9 / Plan 09-06 precedent.
    // For sub-400KB inputs, copy through with a simple resize-to-1170-wide as a safety + metadata strip.
    let pipeline = sharp(srcPath).resize({ width: 1170, withoutEnlargement: true }).png({
      palette: stat.size >= 400 * 1024,
      compressionLevel: 9,
      quality: 80,
    });
    await pipeline.toFile(t.dst);
    const out = fs.statSync(t.dst);
    console.log(t.dst + ": " + Math.round(out.size/1024) + "KB (was " + Math.round(stat.size/1024) + "KB)");
  }
})();
' RAW_DIR="$RAW_DIR"
```

(Per CLAUDE.md "no bundler / build step" — sharp is installed in `/tmp/img-opt`, NOT in the repo's package.json. This matches the Phase 9 / Plan 09-06 pattern verbatim.)
