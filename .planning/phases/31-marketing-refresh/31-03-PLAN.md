**Step 2: Render brand/og-source.svg → og.png using sharp.**

```bash
node -e '
const sharp = require("/tmp/img-opt/node_modules/sharp");
const fs = require("fs");
const svgBuffer = fs.readFileSync("brand/og-source.svg");
sharp(svgBuffer, { density: 144 })  // density 144 = 1.5× DPI; sharp will rasterize at the SVG-declared 1200×630 viewport
  .resize({ width: 1200, height: 630, fit: "fill" })  // explicit lock to 1200×630 even if SVG viewBox drifts
  .png({
    palette: true,           // 256-color palette mode keeps file size small for vector-like compositions
    compressionLevel: 9,     // max zlib compression (Phase 9 / Plan 09-06 pattern)
    quality: 80,             // mozjpeg quality target for any color quantization
  })
  .toFile("og.png")
  .then(info => console.log("og.png rendered:", info.width + "×" + info.height, "size:", Math.round(info.size/1024) + "KB"))
  .catch(err => { console.error("RENDER FAIL:", err.message); process.exit(1); });
' || exit 1
```

**Step 3: Verify the output.**

```bash
node -e '
const sharp = require("/tmp/img-opt/node_modules/sharp");

  console.log("og.png metadata OK: " + m.width + "×" + m.height + " " + m.format);
});
'
ls -la og.png
```
